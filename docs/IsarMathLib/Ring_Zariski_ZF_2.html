section ‹Ring homomorphisms to commutative rings and the Zariski topology›

theory Ring_Zariski_ZF_2 imports Ring_Zariski_ZF_1

begin

text‹In this file we specialize the results about the maps induced on prime spectra
  to ring homomorphisms whose target ring is commutative. In this case the preimage of a prime ideal
  is always a prime ideal, so the induced map on spectra is defined and continuous
  without any assumption on the homomorphism. We also describe the preimages of the
  basic open and closed sets.›

subsection‹The induced map for a commutative target ring›

text‹We work in the context of a ring homomorphism whose target ring is commutative.
  The origin ring does not have to be commutative.›

text‹The locale ‹comm_target_homo› is the ‹ring_homo› locale with the
  additional assumption that the multiplication in the target ring is commutative.›

locale comm_target_homo = ring_homo +
  assumes target_commutative: "V {is commutative on} S"

text‹The induced map on spectra sends a prime ideal of the target ring to its preimage.›

definition (in ring_homo) spec_pullback where
  "spec_pullback ≡ λu∈target_ring.Spec. f-``u"

text‹If the target ring is commutative then the preimage of a prime ideal is a prime ideal.›

lemma (in comm_target_homo) comm_preimage_prime:
  shows "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
  using spectrum_prime_preimage_comm target_commutative by simp

text‹The induced map is a function from the spectrum of the target ring
  to the closed set defined by the kernel in the spectrum of the origin ring.›

lemma (in comm_target_homo) spec_pullback_fun:
  shows "spec_pullback: target_ring.Spec → V⇩S(ker)"
  using spectrum_map comm_preimage_prime unfolding spec_pullback_def by auto

text‹The value of the induced map at a prime ideal is its preimage.›

lemma (in comm_target_homo) spec_pullback_val:
  assumes "u∈target_ring.Spec"
  shows "spec_pullback`u = f-``u"
  using assms beta unfolding spec_pullback_def by auto

text‹The induced map is continuous with respect to the Zariski topologies
  without assuming that the homomorphism is surjective.›

lemma (in comm_target_homo) spec_pullback_cont:
  shows "IsContinuous(τ⇩t, τ⇩o {restricted to}(V⇩S(ker)), spec_pullback)"
  using spectrum_comm_cont target_commutative unfolding spec_pullback_def by auto

subsection‹Preimages of basic sets›

text‹The preimage of the basic open set defined by an element $a$ of the origin ring
  is the basic open set defined by its image $f(a)$.›

lemma (in comm_target_homo) pullback_principal_open:
  assumes "c∈R"
  shows "spec_pullback-``(origin_ring.openBasic({c})) = target_ring.openBasic({f`c})"
proof-
  from assms have sub:"{c} ⊆ R" by auto
  from assms have img:"f``{c} = {f`c}" using func_imagedef f_is_fun by auto
  have "spec_pullback-``(origin_ring.openBasic({c})) = target_ring.openBasic(f``{c})"
    using spectrum_map_vimage_open comm_preimage_prime sub
    unfolding spec_pullback_def by auto
  with img show ?thesis by simp
qed

text‹The preimage of the closed basic set defined by an element $a$ of the origin ring
  is the closed basic set defined by its image $f(a)$.›

lemma (in comm_target_homo) pullback_principal_closed:
  assumes "c∈R"
  shows "spec_pullback-``(origin_ring.closeBasic({c})) = target_ring.closeBasic({f`c})"
proof-
  from assms have sub:"{c} ⊆ R" by auto
  from assms have img:"f``{c} = {f`c}" using func_imagedef f_is_fun by auto
  have "spec_pullback-``(origin_ring.closeBasic({c})) = target_ring.closeBasic(f``{c})"
    using spectrum_map_vimage_closed comm_preimage_prime sub
    unfolding spec_pullback_def by auto
  with img show ?thesis by simp
qed

text‹The preimage of the basic open set defined by a subset $x$ of the origin ring
  is the basic open set defined by the image of $x$.›

lemma (in comm_target_homo) pullback_open:
  assumes "x ⊆ R"
  shows "spec_pullback-``(origin_ring.openBasic(x)) = target_ring.openBasic(f``x)"
  using spectrum_map_vimage_open comm_preimage_prime assms
  unfolding spec_pullback_def by auto

text‹The preimage of the closed basic set defined by a subset $x$ of the origin ring
  is the closed basic set defined by the image of $x$.›

lemma (in comm_target_homo) pullback_closed:
  assumes "x ⊆ R"
  shows "spec_pullback-``(origin_ring.closeBasic(x)) = target_ring.closeBasic(f``x)"
  using spectrum_map_vimage_closed comm_preimage_prime assms
  unfolding spec_pullback_def by auto

subsection‹Homeomorphisms onto the image›

text‹If the ideals of the target ring are extended (see ‹ideals_extended› in
  ‹Ring_Zariski_ZF_1›; this holds for surjective homomorphisms) the induced map is
  a homeomorphism onto its image.›

theorem (in comm_target_homo) spec_pullback_homeomorphism:
  assumes "ideals_extended"
  shows "IsAhomeomorphism(τ⇩t,
    τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}, spec_pullback)"
  using spectrum_map_homeomorphism comm_preimage_prime assms
  unfolding spec_pullback_def by auto

end
