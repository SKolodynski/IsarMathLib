text‹This theory deals with ultrafilters; which is a maximal filter.›

subsection‹Ultrafilters›

text‹In this section we define ultrafilters and prove their basic properties.›

text‹An ultrafilter is a filter that is maximal, i.e. not properly contained in any other filter.›

definition IsUltrafilter ("_{is an ultrafilter on}_" 80)
  where "𝔉{is an ultrafilter on}X ≡ (𝔉{is a filter on}X) ∧ (∀𝔊. (𝔊{is a filter on}X) ⟶ (𝔉 ⊆ 𝔊 ⟶ 𝔊 = 𝔉))"

text‹Every set or its opposite is in an ultrafilter›

lemma set_ultrafilter:
  assumes "A∈Pow(X)" "𝔉{is an ultrafilter on}X"
  shows "A∈𝔉 ∨ (X-A)∈𝔉"
proof-
  from assms(2) have filter:"𝔉{is a filter on}X" unfolding IsUltrafilter_def by auto
  {
    assume a0:"A≠0"
    {
      assume a:"A∉𝔉" "X-A∉𝔉"
      have "A ∈ Pow(X) ⟹ 𝔉 {is a filter on} X ⟹ A ≠ ∅ ⟹
        A ∉ 𝔉 ⟹ (X ∖ A) ∉ 𝔉 ⟹ ∃𝔊. ((𝔊 {is a filter on} X) ∧ A ∈ 𝔊 ∧ 𝔉 ⊆ 𝔊)" using extend_filter by auto
      with a assms(1) a0 filter obtain 𝔊 where g:"𝔊{is a filter on}X" "A ∈ 𝔊" "𝔉 ⊆ 𝔊"
        by auto
      with assms(2) have "A∈𝔉" unfolding IsUltrafilter_def by auto
      with a have False by auto
    }
    then have "A∈𝔉 ∨ (X-A)∈𝔉" by auto
  } moreover
  {
    assume "A=0"
    then have "X-A=X" by auto
    then have "X-A∈𝔉" using filter unfolding IsFilter_def by auto
    then have "A∈𝔉 ∨ (X-A)∈𝔉" by auto
  } ultimately
  show "A∈𝔉 ∨ (X-A)∈𝔉" by blast
qed

text‹It contains only one of them›

lemma only_set_or_opposite:
  assumes "A∈Pow(X)" "𝔉{is a filter on}X" "A∈𝔉"
  shows "(X-A)∉𝔉"
proof
  assume "X-A∈𝔉"
  with assms have "A∩(X-A)∈𝔉" using is_filter_def_split(4) by auto
  moreover have "A∩(X-A) = 0" by auto
  ultimately have "0∈𝔉" by auto
  then show False using assms(2) is_filter_def_split(1) by auto
qed

text‹If a filter contains a set or its opposite, it is in an ultrafilter›

lemma set_ultrafilter_equiv:
  assumes "𝔉{is a filter on}X" "∀A∈Pow(X). A∈𝔉 ∨ (X-A)∈𝔉"
  shows "𝔉{is an ultrafilter on}X"
proof-
  from assms(1) have I:"𝔉{is a filter on}X" by assumption
  {
    fix 𝔊 assume g:"𝔊{is a filter on}X" "𝔉 ⊆ 𝔊"
    {
      assume "𝔊≠𝔉"
      with g(2) obtain h where h:"h:𝔊" "h∉𝔉" by auto
      from h(1) g(1) have "h∈Pow(X)" "h≠0" unfolding IsFilter_def by auto
      with assms(2) h(2) have "X-h∈𝔉" by auto
      with g(2) have "X-h:𝔊" by auto
      with h(1) g(1) have "(X-h)∩h∈𝔊" unfolding IsFilter_def by auto moreover
      have "(X-h)∩h=0" by auto ultimately
      have "0∈𝔊" by auto
      with g(1) have "False" unfolding IsFilter_def by auto
    }
    then have "𝔊=𝔉" by auto
  }
  then have II:"∀𝔊. (𝔊 {is a filter on} X) ⟶ 𝔉 ⊆ 𝔊 ⟶ 𝔊 =𝔉" by auto
  from I II show ?thesis unfolding IsUltrafilter_def by auto
qed

text‹Filters of containing a point, are ultrafilters›

lemma include_point_ultrafilter:
  assumes "x∈X"
  shows "{A∈Pow(X). x∈A}{is an ultrafilter on}X"
proof-
  have I:"0∉{A ∈ Pow(X) . x ∈ A}" by auto
  from assms have II:"X∈{A ∈ Pow(X) . x ∈ A}" by auto
  have III:"{A ∈ Pow(X) . x ∈ A} ⊆ Pow(X)" by auto
  {
    fix A B assume AB:"A∈{A ∈ Pow(X) . x ∈ A}" "B∈{A ∈ Pow(X) . x ∈ A}"
    then have "A∩B∈{A ∈ Pow(X) . x ∈ A}" by auto
  }
  then have IV:"∀A∈{A ∈ Pow(X) . x ∈ A}. ∀B∈{A ∈ Pow(X) . x ∈ A}. A∩B∈{A ∈ Pow(X) . x ∈ A}" by auto
  {
    fix B C assume B:"B∈{A ∈ Pow(X) . x ∈ A}" "C∈Pow(X)" "B⊆C"
    then have "C∈{A ∈ Pow(X) . x ∈ A}" by auto
  }
  then have V:"∀B∈{A ∈ Pow(X) . x ∈ A}. ∀C∈Pow(X). B ⊆ C ⟶ C∈{A ∈ Pow(X) . x ∈ A}" by auto
  from I II III IV V
  have I:"{A ∈ Pow(X) . x ∈ A} {is a filter on} X" unfolding IsFilter_def by blast
  {
    fix A assume A:"A∈Pow(X)"
    have "x:A ∨ x∉A" by auto
    with assms have "x∈A ∨ x∈X-A" by auto
    with A(1) have "A∈{A ∈ Pow(X) . x ∈ A} ∨ X-A: {A ∈ Pow(X) . x ∈ A} " by auto
  }
  then have II:"∀A∈Pow(X). A ∈ {A ∈ Pow(X) . x ∈ A} ∨ X - A ∈ {A ∈ Pow(X) . x ∈ A}" by auto
  from I II show ?thesis using set_ultrafilter_equiv by blast
qed

text‹An ultrafilter that has no singleton sets, does not have finite sets›

lemma ultrafilter_finite_set:
  assumes "𝔉{is an ultrafilter on}X" "∀x∈X. {x}∉𝔉" "A∈FinPow(X)"
  shows "A∉𝔉"
proof-
  from assms(1) have I:"0∉𝔉" unfolding IsUltrafilter_def IsFilter_def by auto
  have III:"A∈FinPow(X)" using assms(3) by assumption
  {
    fix B assume b:"B∈FinPow(X)" "B∉𝔉"
    {
      fix s assume s:"s∈X"
      {
        assume cont:"B∪{s}∈𝔉"
        {
          assume "B=0"
          then have "B∪{s} = {s}" by auto
          with s assms(2) cont have False by auto
        } moreover
        {
          assume "B≠0"
          with b have "X-B ∈𝔉" using assms(1) set_ultrafilter
            unfolding FinPow_def by auto moreover
          have "{s} ∈ Pow(X) ⟹ 𝔉{is an ultrafilter on}X ⟹
            {s} ∈ 𝔉 ∨ X ∖ {s} ∈ 𝔉" using set_ultrafilter by auto
          with assms(1,2) s have "X-{s}∈𝔉" by auto
          ultimately have "(X-B)∩(X-{s}) ∈𝔉" using assms(1)
            unfolding IsUltrafilter_def IsFilter_def by auto moreover
          {
            fix q assume "q∈(X-B)∩(X-{s})"
            then have "q:X" "q∉B" "q∉{s}" by auto
            then have "q∈X-(B∪{s})" by auto
          }
          then have "(X-B)∩(X-{s}) ⊆ X-(B∪{s})" by auto moreover
          have "X-(B∪{s})∈Pow(X)" by auto
          ultimately have "X-(B∪{s}) ∈𝔉" using assms(1) unfolding IsFilter_def IsUltrafilter_def
            by blast
          with cont have "(X-(B∪{s}))∩(B∪{s}):  𝔉" using assms(1)
            unfolding IsFilter_def IsUltrafilter_def by auto moreover
          {
            fix q assume "q∈(X-(B∪{s}))∩(B∪{s})"
            then have False by auto
          }
          then have "(X-(B∪{s}))∩(B∪{s}) = 0" by auto
          ultimately have "0:  𝔉" by auto
          with assms(1) have "False" unfolding IsUltrafilter_def IsFilter_def by auto
        }
        ultimately have "False" by auto
      }
      then have "B∪{s}∉𝔉" by auto
    }
    then have "∀q∈X. B ∪ {q} ∉ 𝔉" by auto
  }
  then have II:"∀A∈FinPow(X). A ∉ 𝔉 ⟶ (∀q∈X. A ∪ {q} ∉ 𝔉)" by auto
  from I II III show ?thesis by (rule FinPow_induct)
qed

text‹The cofinite filter plays an important role in ultrafilters.›

corollary ultrafilter_contains_cofinite:
  assumes "𝔉{is an ultrafilter on}X" "∀x∈X. {x}∉𝔉"
  shows "(CoFinite(X))-{0} ⊆ 𝔉"
proof
  fix S assume "S∈(CoFinite(X))-{0}"
  then have s:"S ∈ Pow(X)" "X - S ≺ nat" "S≠0" unfolding CoCardinal_def Cofinite_def by auto
  then have "Finite(X-S)" using lesspoll_nat_is_Finite by auto
  then have "X-S∈FinPow(X)" unfolding FinPow_def by auto
  with ultrafilter_finite_set assms have "X-S∉𝔉" by auto
  with assms(1) show "S∈𝔉" using set_ultrafilter s(1,3) by auto
qed

text‹An ultrafilter is free or principal.›

corollary ultrafilter_principal_or_free:
  assumes "𝔉{is an ultrafilter on}X"
  shows "⋂𝔉 = 0 ∨ (∃x∈X. ⋂𝔉 = {x})"
proof-
  {
    assume "¬(∃x∈X. ⋂𝔉 = {x})"
    then have r:"∀x∈X. ⋂𝔉≠{x}" by auto
    from assms have a:"X≠0" unfolding IsUltrafilter_def IsFilter_def by auto
    {
      fix x assume x:"x:X" "{x}:𝔉"
      from this(2) assms have "∀C∈Pow(X). {x} ⊆ C ⟶ C ∈ 𝔉" 
        unfolding IsFilter_def IsUltrafilter_def by auto
      then have s:"{C∈Pow(X). x∈C} ⊆ 𝔉" by auto
      {
        fix q assume q:"q∈𝔉" "x∉q"
        from q(2) s x(1) have "X-q:𝔉" by auto
        with q(1) have "(X-q)∩q∈𝔉" using assms unfolding IsUltrafilter_def IsFilter_def by auto
        moreover have "(X-q)∩q = 0" by auto
        ultimately have "0:𝔉" by auto
        with assms have "False" unfolding IsUltrafilter_def IsFilter_def by auto
      }
      then have "𝔉⊆{C∈Pow(X). x∈C}" using assms unfolding IsUltrafilter_def IsFilter_def by auto
      with s have "𝔉={C∈Pow(X). x∈C}" by auto
      then have "⋂𝔉={x}" using x(2) by auto
      with r x(1) have "False" by auto
    }
    then have "∀x∈X. {x}∉𝔉" by auto
    then have sub:"(Cofinite(X))-{0} ⊆ 𝔉" using ultrafilter_contains_cofinite assms by auto
    have "(Cofinite(X)){is a topology}" using CoCar_is_topology InfCard_nat unfolding Cofinite_def by auto
    then have "⋃Cofinite(X)∈Cofinite(X)" using carr_open by blast
    then have b:"X ∈(Cofinite(X))" using
        union_cocardinal unfolding Cofinite_def by auto
    from a b have "X∈(Cofinite(X))-{0}" by auto
    with sub have A:"⋂𝔉 ⊆ ⋂((Cofinite(X))-{0})" by auto
    {
      fix q assume q:"q∈⋂((Cofinite(X))-{0})"
      from a b have B:"⋂((Cofinite(X))-{0}) ⊆ X" by auto
      with q have qX:"q∈X" by auto
      then have "X-(X-{q}) = {q}" by auto
      then have "X-(X-{q})≈1" using singleton_eqpoll_1 by auto
      then have "X-(X-{q}) ≺ nat" using eq_lesspoll_trans n_lesspoll_nat nat_1I by auto
      then have cof:"X-{q}∈(Cofinite(X))" unfolding Cofinite_def CoCardinal_def by auto
      {
        assume "X-{q} = 0"
        with qX have "X={q}" by auto
        then have "⋂𝔉 ⊆ {q}"  using assms unfolding IsUltrafilter_def IsFilter_def by auto
        with qX r have "⋂𝔉 = 0" by blast
      } moreover
      {
        assume "X-{q} ≠ 0"
        with cof have "X-{q}∈(Cofinite(X)) -{0}" by auto
        then have "⋂((Cofinite(X))-{0}) ⊆ X-{q}" by auto
        with q have False by auto
        then have "⋂𝔉 = 0" by auto
      } ultimately
      have "⋂𝔉 = 0" by auto
    }
    then have "⋂𝔉=0 ∨ ⋂((Cofinite(X))-{0}) = 0" by blast
    with A have "⋂𝔉=0" by auto
  }
  then show ?thesis by auto
qed
