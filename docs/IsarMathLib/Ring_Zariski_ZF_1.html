section ‹Ring homomorphisms as continuous maps on Spec›

theory Ring_Zariski_ZF_1 imports Ring_Zariski_ZF Ring_ZF_3 Topology_ZF_2

begin

text‹Every ring homomorphism $f: R \to S$ induces a map $f^*: \mathrm{Spec}(S) \to \mathrm{Spec}(R)$
  on prime spectra by pulling back prime ideals. This file shows that $f^*$ is continuous with
  respect to the Zariski topologies on $\mathrm{Spec}(R)$ and $\mathrm{Spec}(S)$, establishing
  $\mathrm{Spec}$ as a contravariant functor from commutative rings to topological spaces.›

subsection‹Induced maps on spectra›

text‹An element of the spectrum of the target ring is a prime ideal, an ideal, and a subset of $S$.›

lemma (in ring_homo) target_spec_dest:
  assumes "u∈target_ring.Spec"
  shows "u◃⇩pR⇩t" "u◃R⇩t" "u⊆S"
proof-
  from assms show p:"u◃⇩pR⇩t" unfolding target_ring.Spec_def by auto
  from p show i:"u◃R⇩t" using target_ring.primeIdeal_def by auto
  from i show "u⊆S" using target_ring.ideal_dest_subset by auto
qed

text‹If preimages of primes are prime then the preimage of every prime ideal of the target ring
  contains the kernel, hence it belongs to the V-set of the kernel.›

lemma (in ring_homo) spectrum_prime_preimage_kernel:
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
  shows "∀u∈target_ring.Spec. f-``u ∈ V⇩S(ker)"
proof
  fix u assume u:"u∈target_ring.Spec"
  have sub:"f-``{𝟬⇩S} ⊆ R" using func1_1_L3 f_is_fun by auto
  then have clo:"origin_ring.closeBasic(f-``{𝟬⇩S}) = {K∈origin_ring.Spec. f-``{𝟬⇩S} ⊆ K}"
    using origin_ring.closeBasic_def by auto
  from u have "u◃R⇩t" using target_spec_dest by auto
  then have "𝟬⇩S ∈ u" using target_ring.ideal_dest_zero by auto
  then have "{𝟬⇩S} ⊆ u" by auto
  then have "f-``{𝟬⇩S} ⊆ f-``u" by auto
  moreover from assms u have "f-``u ∈ origin_ring.Spec" by auto
  ultimately show "f-``u ∈ origin_ring.closeBasic(f-``{𝟬⇩S})" using clo by auto
qed

text‹If the preimage of every prime ideal of the target ring is a prime ideal of the origin ring,
  then pulling back prime ideals defines a map between the prime spectra
  that lands in the closed set defined by the kernel.›

lemma (in ring_homo) spectrum_map:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
  shows "g: target_ring.Spec → V⇩S(ker)"
proof-
  have "g: target_ring.Spec → {f-``u. u∈target_ring.Spec}" using lam_funtype
    unfolding g_def by auto moreover
  have "{f-``u. u∈target_ring.Spec} ⊆ V⇩S(ker)"
    using spectrum_prime_preimage_kernel assms(2) by auto
  ultimately show ?thesis using func1_1_L1B by auto
qed

text‹The hypothesis of ‹spectrum_map› holds when the homomorphism is surjective.›

lemma (in ring_homo) spectrum_prime_preimage_surj:
  assumes "f∈surj(R,S)"
  shows "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
proof
  fix u assume "u∈target_ring.Spec"
  then have "u◃⇩pR⇩t" unfolding target_ring.Spec_def by auto
  with assms have "(f-``u)◃⇩pR⇩o" using preimage_prime_ideal_surj by auto
  then show "f-``u ∈ origin_ring.Spec" unfolding origin_ring.Spec_def
    origin_ring.primeIdeal_def using origin_ring.ideal_dest_subset by auto
qed

text‹The hypothesis of ‹spectrum_map› also holds when the target ring is commutative.›

lemma (in ring_homo) spectrum_prime_preimage_comm:
  assumes "V {is commutative on} S"
  shows "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
proof
  fix u assume "u∈target_ring.Spec"
  then have "u◃⇩pR⇩t" unfolding target_ring.Spec_def by auto
  with assms have "(f-``u)◃⇩pR⇩o" using preimage_prime_ideal_comm by auto
  then show "f-``u ∈ origin_ring.Spec" unfolding origin_ring.Spec_def
    origin_ring.primeIdeal_def using origin_ring.ideal_dest_subset by auto
qed

text‹A surjective ring homomorphism induces a map between the prime spectra that lands in the closed set defined by the kernel.›

lemma (in ring_homo) spectrum_surj:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "f∈surj(R,S)"
  shows "g: target_ring.Spec → V⇩S(ker)"
  using spectrum_map spectrum_prime_preimage_surj assms(2)
  unfolding g_def by auto

text‹The induced map on spectra is a bijection onto the V-set of the kernel.›

lemma (in ring_homo) spectrum_surj_bij:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "f∈surj(R,S)"
  shows "g∈bij(target_ring.Spec, V⇩S(ker))"
proof-
  {
    fix s t assume st:"s∈target_ring.Spec" "t∈target_ring.Spec"
      "g`s = g`t"
    then have "f-``s = f-``t" using beta unfolding g_def by auto
    then have "f``(f-``s) = f``(f-``t)" by auto
    moreover from st(1,2) have "s ⊆ S" "t ⊆S"
      unfolding target_ring.Spec_def origin_ring.Spec_def
      by auto
    moreover note assms(2) st(1,2)
    ultimately have "s=t" using surj_image_vimage
      unfolding target_ring.Spec_def origin_ring.Spec_def
      by auto
  }
  then have "g∈inj(target_ring.Spec, origin_ring.closeBasic(f-``{𝟬⇩S}))"
    unfolding inj_def using spectrum_surj assms(2) unfolding g_def by auto
  moreover
  {
    fix t assume t:"t∈origin_ring.closeBasic(f-``{𝟬⇩S})"
    then have tt:"t∈origin_ring.Spec" "f-``{𝟬⇩S} ⊆ t"
      using origin_ring.closeBasic_def func1_1_L6A[OF surj_is_fun]
      assms(2) by auto
    {
      fix y assume y:"y∈f-``(f``t)"
      then have y:"y∈R" "f`y∈f``t" using func1_1_L15
        surj_is_fun[OF assms(2)] by auto
      from y(2) obtain x where x:"x∈t" "f`y = f`x"
        using func_imagedef[OF surj_is_fun]
        assms(2) tt(1) unfolding origin_ring.Spec_def by auto
      from x(2) have "(f`y)\<rs>⇩S(f`x) = (f`x)\<rs>⇩S(f`x)" by auto
      then have "(f`y)\<rs>⇩S(f`x) = 𝟬⇩S" using target_ring.Ring_ZF_1_L3(7)
        apply_type[OF surj_is_fun] assms(2) x(1) tt(1)
        unfolding origin_ring.Spec_def by auto
      then have "f`(y\<rs>⇩Rx) = 𝟬⇩S" using homomor_dest_subs
        x(1) tt(1) y(1) unfolding origin_ring.Spec_def by auto moreover
      from x(1) tt(1) have "x∈R" unfolding origin_ring.Spec_def by auto
      with y(1) have "y\<rs>⇩Rx ∈R" using origin_ring.Ring_ZF_1_L4(2) by auto
      ultimately have "y\<rs>⇩Rx ∈ f-``{𝟬⇩S}" using func1_1_L15
        surj_is_fun[OF assms(2)] by auto
      then have "y\<rs>⇩Rx ∈t" using tt(2) by auto moreover
      have "t◃R⇩o" using tt(1) unfolding origin_ring.Spec_def by auto
      ultimately have "x\<ra>⇩R(y\<rs>⇩Rx) ∈t" using x(1)
        origin_ring.ideal_dest_sum by auto
      then have "y∈t" using origin_ring.Ring_ZF_2_L1A(5)
        y(1) ‹x∈R› by auto
    }
    then have "f-``(f``t) = t" using func1_1_L9[of f R S t]
      surj_is_fun[OF assms(2)] tt(1) unfolding origin_ring.Spec_def
      by auto
    from tt(1) have "t◃⇩pR⇩o" unfolding origin_ring.Spec_def by auto
    with tt(2) assms(2) have "(f``t)◃⇩pR⇩t"
      using preimage_ideal_prime by auto
    then have "(f``t)◃⇩pR⇩t" "(f``t)◃R⇩t"
      using target_ring.primeIdeal_def[of "f``t"]
      by auto
    then have "(f``t):target_ring.Spec"
      unfolding target_ring.Spec_def using target_ring.ideal_dest_subset
      by auto
    then have "g`(f``t) = f-``(f``t)" unfolding g_def
      using beta by auto
    with ‹f-``(f``t) = t› have "g`(f``t) = t" by auto
    with ‹(f``t):target_ring.Spec› have "∃p∈target_ring.Spec. g`p = t" by auto
  }
  ultimately show "g:bij(target_ring.Spec, V⇩S(ker))"
    unfolding bij_def surj_def inj_def by auto
qed

subsection‹Continuity of the induced map›

text‹We define the Zariski topologies on origin and target spectra, then show the induced map is continuous and open.›

text‹The Zariski topology on the origin ring's spectrum.›

definition (in ring_homo) top_origin ("τ⇩o") where
  "top_origin ≡ {origin_ring.openBasic(J) . J ∈ origin_ring.ideals}"

text‹The Zariski topology on the target ring's spectrum.›

definition (in ring_homo) top_target ("τ⇩t") where
  "top_target ≡ {target_ring.openBasic(J) . J ∈ target_ring.ideals}"

text‹Continuity of a map between spectra with respect to the Zariski topologies.›

definition (in ring_homo) spec_cont where
 "spec_cont(h) ≡ IsContinuous(τ⇩t, τ⇩o, h)"

text‹For subsets $x\subseteq R$ and $t\subseteq S$ the image of $x$ is contained in $t$
  iff $x$ is contained in the preimage of $t$. This holds for any function.›

lemma (in ring_homo) image_subset_iff_subset_vimage:
  assumes "x ⊆ R" "t ⊆ S"
  shows "f``x ⊆ t ⟷ x ⊆ f-``t"
proof
  assume as:"f``x ⊆ t"
  { fix y assume y:"y∈x"
    with assms(1) have "y∈R" by auto
    have "f``x = {f`z. z∈x}" using func_imagedef f_is_fun assms(1) by auto
    with y have "f`y ∈ f``x" by auto
    with as have "f`y ∈ t" by auto
    with ‹y∈R› have "y ∈ f-``t" using func1_1_L15 f_is_fun by auto
  }
  then show "x ⊆ f-``t" by auto
next
  assume as:"x ⊆ f-``t"
  { fix s assume "s∈f``x"
    moreover have "f``x = {f`z. z∈x}" using func_imagedef f_is_fun assms(1) by auto
    ultimately obtain y where y:"y∈x" "s=f`y" by auto
    with as have "f`y ∈ t" using func1_1_L15 f_is_fun by auto
    with y(2) have "s∈t" by auto
  }
  then show "f``x ⊆ t" by auto
qed

text‹The preimage under the induced map of the basic open set ‹D(x)› of the origin spectrum
  is the basic open set ‹D(f``x)› of the target spectrum. This only requires that the induced
  map is well defined, not that the homomorphism is surjective.›

lemma (in ring_homo) spectrum_map_vimage_open:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec" "x ⊆ R"
  shows "g-``(origin_ring.openBasic(x)) = target_ring.openBasic(f``x)"
proof-
  have gfun:"g: target_ring.Spec → V⇩S(ker)"
    using spectrum_map assms(2) unfolding g_def by auto
  have fx:"f``x ⊆ S" using func1_1_L6(2) f_is_fun by auto
  have key:"∀t∈target_ring.Spec. g`t ∈ origin_ring.openBasic(x) ⟷
    t ∈ target_ring.openBasic(f``x)"
  proof
    fix t assume t:"t∈target_ring.Spec"
    from t have tS:"t⊆S" using target_spec_dest by auto
    from t have gt:"g`t = f-``t" unfolding g_def using beta by auto
    from t assms(2) have tp:"f-``t ∈ origin_ring.Spec" by auto
    have a:"f-``t ∈ origin_ring.openBasic(x) ⟷ ¬(x ⊆ f-``t)"
      using tp origin_ring.openBasic_def assms(3) by auto
    have b:"t ∈ target_ring.openBasic(f``x) ⟷ ¬(f``x ⊆ t)"
      using t target_ring.openBasic_def fx by auto
    have c:"f``x ⊆ t ⟷ x ⊆ f-``t"
      using image_subset_iff_subset_vimage assms(3) tS by auto
    from a b c gt show "g`t ∈ origin_ring.openBasic(x) ⟷
      t ∈ target_ring.openBasic(f``x)" by auto
  qed
  show ?thesis
  proof
    { fix t assume "t ∈ g-``(origin_ring.openBasic(x))"
      then have "t∈target_ring.Spec" "g`t ∈ origin_ring.openBasic(x)"
        using func1_1_L15 gfun by auto
      with key have "t ∈ target_ring.openBasic(f``x)" by auto
    }
    then show "g-``(origin_ring.openBasic(x)) ⊆ target_ring.openBasic(f``x)" by auto
    { fix t assume t:"t ∈ target_ring.openBasic(f``x)"
      then have ts:"t∈target_ring.Spec" using target_ring.openBasic_def fx by auto
      with key t have "g`t ∈ origin_ring.openBasic(x)" by auto
      with ts have "t ∈ g-``(origin_ring.openBasic(x))" using func1_1_L15 gfun by auto
    }
    then show "target_ring.openBasic(f``x) ⊆ g-``(origin_ring.openBasic(x))" by auto
  qed
qed

text‹The preimage of a closed basic set ‹V(x)› of the origin spectrum is the closed basic set
  ‹V(f``x)› of the target spectrum.›

lemma (in ring_homo) spectrum_map_vimage_closed:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec" "x ⊆ R"
  shows "g-``(origin_ring.closeBasic(x)) = target_ring.closeBasic(f``x)"
proof-
  have gfun:"g: target_ring.Spec → V⇩S(ker)"
    using spectrum_map assms(2) unfolding g_def by auto
  have fx:"f``x ⊆ S" using func1_1_L6(2) f_is_fun by auto
  have key:"∀t∈target_ring.Spec. g`t ∈ origin_ring.closeBasic(x) ⟷
    t ∈ target_ring.closeBasic(f``x)"
  proof
    fix t assume t:"t∈target_ring.Spec"
    from t have tS:"t⊆S" using target_spec_dest by auto
    from t have gt:"g`t = f-``t" unfolding g_def using beta by auto
    from t assms(2) have tp:"f-``t ∈ origin_ring.Spec" by auto
    have a:"f-``t ∈ origin_ring.closeBasic(x) ⟷ x ⊆ f-``t"
      using tp origin_ring.closeBasic_def assms(3) by auto
    have b:"t ∈ target_ring.closeBasic(f``x) ⟷ f``x ⊆ t"
      using t target_ring.closeBasic_def fx by auto
    have c:"f``x ⊆ t ⟷ x ⊆ f-``t"
      using image_subset_iff_subset_vimage assms(3) tS by auto
    from a b c gt show "g`t ∈ origin_ring.closeBasic(x) ⟷
      t ∈ target_ring.closeBasic(f``x)" by auto
  qed
  show ?thesis
  proof
    { fix t assume "t ∈ g-``(origin_ring.closeBasic(x))"
      then have "t∈target_ring.Spec" "g`t ∈ origin_ring.closeBasic(x)"
        using func1_1_L15 gfun by auto
      with key have "t ∈ target_ring.closeBasic(f``x)" by auto
    }
    then show "g-``(origin_ring.closeBasic(x)) ⊆ target_ring.closeBasic(f``x)" by auto
    { fix t assume t:"t ∈ target_ring.closeBasic(f``x)"
      then have ts:"t∈target_ring.Spec" using target_ring.closeBasic_def fx by auto
      with key t have "g`t ∈ origin_ring.closeBasic(x)" by auto
      with ts have "t ∈ g-``(origin_ring.closeBasic(x))" using func1_1_L15 gfun by auto
    }
    then show "target_ring.closeBasic(f``x) ⊆ g-``(origin_ring.closeBasic(x))" by auto
  qed
qed

text‹The induced map on spectra is continuous with respect to the Zariski topologies,
  where the topology of the origin spectrum is restricted to any set ‹W› containing
  the image of the map. This holds whenever the preimages of prime ideals are prime ideals,
  in particular the homomorphism does not have to be surjective.›

lemma (in ring_homo) spectrum_map_cont:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
    "∀u∈target_ring.Spec. f-``u ∈ W"
  shows "IsContinuous(τ⇩t, τ⇩o {restricted to} W, g)"
  unfolding IsContinuous_def top_target_def RestrictedTo_def top_origin_def
proof(safe)
  fix x assume ass:"x◃R⇩o" "x ⊆ R"
  have gfun:"g: target_ring.Spec → W"
  proof-
    have "g: target_ring.Spec → {f-``u. u∈target_ring.Spec}" using lam_funtype
      unfolding g_def by auto
    moreover have "{f-``u. u∈target_ring.Spec} ⊆ W" using assms(3) by auto
    ultimately show ?thesis using func1_1_L1B by auto
  qed
  have vo:"g-``(origin_ring.openBasic(x)) = target_ring.openBasic(f``x)"
    using spectrum_map_vimage_open assms(2) ass(2) unfolding g_def by auto
  have fx:"f``x ⊆ S" using func1_1_L6(2) f_is_fun by auto
  have eq:"g-``(W ∩ origin_ring.openBasic(x)) = target_ring.openBasic(f``x)"
  proof
    { fix t assume "t ∈ g-``(W ∩ origin_ring.openBasic(x))"
      then have "t ∈ g-``(origin_ring.openBasic(x))" by auto
      with vo have "t ∈ target_ring.openBasic(f``x)" by auto
    }
    then show "g-``(W ∩ origin_ring.openBasic(x)) ⊆ target_ring.openBasic(f``x)" by auto
    { fix t assume "t ∈ target_ring.openBasic(f``x)"
      with vo have t:"t ∈ g-``(origin_ring.openBasic(x))" by auto
      then have ts:"t∈target_ring.Spec" "g`t ∈ origin_ring.openBasic(x)"
        using func1_1_L15 gfun by auto
      from ts(1) gfun have "g`t ∈ W" using apply_type by auto
      with ts(2) have "g`t ∈ W ∩ origin_ring.openBasic(x)" by auto
      with ts(1) have "t ∈ g-``(W ∩ origin_ring.openBasic(x))"
        using func1_1_L15 gfun by auto
    }
    then show "target_ring.openBasic(f``x) ⊆ g-``(W ∩ origin_ring.openBasic(x))" by auto
  qed
  have "∃J∈target_ring.ideals. target_ring.openBasic(f``x) = target_ring.openBasic(J)"
    using target_ring.D_operator_only_ideals[of "f``x"] fx 
      target_ring.generated_ideal_is_ideal[of "f``x"]
      target_ring.ideal_dest_subset[of "target_ring.generatedIdeal(f `` x)"] by auto
  then obtain J where "J∈target_ring.ideals" "target_ring.openBasic(f``x) = target_ring.openBasic(J)"
    by auto
  with eq show "g -`` (W ∩ origin_ring.openBasic(x)) ∈
           RepFun(target_ring.ideals, target_ring.openBasic)"
    by auto
qed

text‹For a surjective homomorphism the induced map on spectra is continuous.›

lemma (in ring_homo) spectrum_surj_cont:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "f∈surj(R,S)"
  shows "IsContinuous(τ⇩t, τ⇩o {restricted to}(V⇩S(ker)), g)"
proof-
  have h:"∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
    using spectrum_prime_preimage_surj assms(2) by auto
  then have k:"∀u∈target_ring.Spec. f-``u ∈ V⇩S(ker)"
    using spectrum_prime_preimage_kernel by auto
  from h k show ?thesis using spectrum_map_cont unfolding g_def by auto
qed

text‹The same holds if the target ring is commutative, without assuming surjectivity.›

lemma (in ring_homo) spectrum_comm_cont:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "V {is commutative on} S"
  shows "IsContinuous(τ⇩t, τ⇩o {restricted to}(V⇩S(ker)), g)"
proof-
  have h:"∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
    using spectrum_prime_preimage_comm assms(2) by auto
  then have k:"∀u∈target_ring.Spec. f-``u ∈ V⇩S(ker)"
    using spectrum_prime_preimage_kernel by auto
  from h k show ?thesis using spectrum_map_cont unfolding g_def by auto
qed

subsection‹Openness and homeomorphisms without surjectivity›

text‹The induced map on spectra is open and injective, hence a homeomorphism onto its image,
  under a condition on the ideals of the target ring that is implied by surjectivity but
  also holds in other cases.›

text‹The ideals of the target ring of a surjective homomorphism are extended.›

lemma (in ring_homo) ideals_extended_surj:
  assumes "f∈surj(R,S)"
  shows "ideals_extended"
  unfolding IdealsExtended_def
proof(intro allI impI)
  fix J K assume J:"J◃R⇩t" and Kc:"K◃R⇩t ∧ f``(f-``J) ⊆ K"
  from J have "J ⊆ S" using target_ring.ideal_dest_subset by auto
  with assms have "f``(f-``J) = J" using surj_image_vimage by auto
  with Kc show "J ⊆ K" by auto
qed

text‹If the ideals of the target ring are extended, the induced map on spectra is injective
  and hence a bijection onto its image.›

lemma (in ring_homo) spectrum_map_bij:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "ideals_extended"
  shows "g ∈ bij(target_ring.Spec, {f-``u. u∈target_ring.Spec})"
proof-
  have gfn:"g: target_ring.Spec → {f-``u. u∈target_ring.Spec}" using lam_funtype
    unfolding g_def by auto
  have sub:"∀u∈target_ring.Spec. f-``u ⊆ R" using func1_1_L3 f_is_fun by auto
  { fix s t assume st:"s∈target_ring.Spec" "t∈target_ring.Spec" "g`s = g`t"
    then have eq:"f-``s = f-``t" using beta unfolding g_def by auto
    from st(1,2) have ss:"s◃R⇩t" "s⊆S" and tt:"t◃R⇩t" "t⊆S"
      using target_spec_dest by auto
    from st(1,2) sub have rs:"f-``s ⊆ R" "f-``t ⊆ R" by auto
    have "f``(f-``s) ⊆ t ⟷ f-``s ⊆ f-``t"
      using image_subset_iff_subset_vimage rs(1) tt(2) by auto
    with eq have "f``(f-``s) ⊆ t" by auto
    with assms(2) ss(1) tt(1) have "s ⊆ t" using ideals_extended_dest by auto
    moreover
    have "f``(f-``t) ⊆ s ⟷ f-``t ⊆ f-``s"
      using image_subset_iff_subset_vimage rs(2) ss(2) by auto
    with eq have "f``(f-``t) ⊆ s" by auto
    with assms(2) ss(1) tt(1) have "t ⊆ s" using ideals_extended_dest by auto
    ultimately have "s = t" by auto
  }
  then have "g ∈ inj(target_ring.Spec, {f-``u. u∈target_ring.Spec})"
    unfolding inj_def using gfn by auto
  moreover
  { fix y assume "y ∈ {f-``u. u∈target_ring.Spec}"
    then obtain u where u:"u∈target_ring.Spec" "y = f-``u" by auto
    then have "g`u = y" unfolding g_def using beta by auto
    with u(1) have "∃x∈target_ring.Spec. g`x = y" by auto
  }
  then have "g ∈ surj(target_ring.Spec, {f-``u. u∈target_ring.Spec})"
    unfolding surj_def using gfn by auto
  ultimately show ?thesis unfolding bij_def by auto
qed

text‹If the ideals of the target ring are extended, the induced map on spectra sends open sets
  to open sets of the topology of the origin spectrum restricted to the image of the map.
  Surjectivity of the homomorphism is not needed.›

lemma (in ring_homo) spectrum_map_open:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec" "ideals_extended"
  shows "∀U∈τ⇩t. g``U ∈ τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}"
proof
  fix U assume U:"U∈τ⇩t"
  then obtain I where I:"I◃R⇩t" "I⊆S"
    "U=target_ring.openBasic(I)" unfolding top_target_def
    by auto
  from I(3) have sub:"U ⊆ target_ring.Spec"
    using target_ring.openBasic_def I(2) by auto
  have gfun:"g: target_ring.Spec → V⇩S(ker)"
    using spectrum_map assms(2) unfolding g_def by auto
  have gU:"g``U = {g`u. u∈U}" using func_imagedef gfun sub by auto
  have fI:"f-``I ⊆ R" using func1_1_L3 f_is_fun by auto
  have eq:"g``U = {f-``u. u∈target_ring.Spec} ∩ origin_ring.openBasic(f-``I)"
  proof
    { fix s assume "s∈g``U"
      then obtain u where u:"u∈U" "s=g`u" using gU by auto
      with sub have us:"u∈target_ring.Spec" by auto
      with u(2) have s:"s=f-``u" unfolding g_def using beta by auto
      from us assms(2) have sp:"f-``u ∈ origin_ring.Spec" by auto
      from us have uI:"u◃R⇩t" "u⊆S" using target_spec_dest by auto
      from u(1) I(2,3) have Iu:"¬(I ⊆ u)" using target_ring.openBasic_def
        by auto
      have ii:"f``(f-``I) ⊆ u ⟷ f-``I ⊆ f-``u"
        using image_subset_iff_subset_vimage fI uI(2) by auto
      { assume "f-``I ⊆ f-``u"
        with ii have "f``(f-``I) ⊆ u" by auto
        with assms(3) I(1) uI(1) have "I ⊆ u" using ideals_extended_dest by auto
        with Iu have False by auto
      }
      then have "¬(f-``I ⊆ s)" using s by auto
      with sp s have "s ∈ origin_ring.openBasic(f-``I)"
        using origin_ring.openBasic_def fI by auto
      moreover from us s have "s ∈ {f-``u. u∈target_ring.Spec}" by auto
      ultimately have "s ∈ {f-``u. u∈target_ring.Spec} ∩ origin_ring.openBasic(f-``I)"
        by auto
    }
    then show "g``U ⊆ {f-``u. u∈target_ring.Spec} ∩ origin_ring.openBasic(f-``I)"
      by auto
    { fix s assume s:"s ∈ {f-``u. u∈target_ring.Spec} ∩ origin_ring.openBasic(f-``I)"
      then obtain u where u:"u∈target_ring.Spec" "s=f-``u" by auto
      from s have "s ∈ origin_ring.openBasic(f-``I)" by auto
      with fI have nsub:"¬(f-``I ⊆ s)" using origin_ring.openBasic_def by auto
      { assume "I ⊆ u"
        then have "f-``I ⊆ f-``u" by auto
        with u(2) nsub have False by auto
      }
      then have "¬(I ⊆ u)" by auto
      with u(1) have "u ∈ U" using I(3) target_ring.openBasic_def I(2) by auto
      then have "g`u∈{g`u. u∈U}" by auto
      moreover from u(1) have "g`u = f-``u" unfolding g_def using beta by auto
      ultimately have "f-``u∈g``U" using gU by auto
      then have "s ∈ g``U" using u(2) by blast
    }
    then show "{f-``u. u∈target_ring.Spec} ∩ origin_ring.openBasic(f-``I) ⊆ g``U"
      by auto
  qed
  from I(1) have "(f-``I) ◃R⇩o" using preimage_ideal(2) by auto
  then have "f-``I ∈ origin_ring.ideals" using origin_ring.ideal_dest_subset by auto
  then have "origin_ring.openBasic(f-``I) ∈ τ⇩o" unfolding top_origin_def by auto
  with eq show "g``U ∈ τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}"
    unfolding RestrictedTo_def by auto
qed

text‹If the preimages of prime ideals are prime and the ideals of the target ring are extended,
  then the induced map on spectra is a homeomorphism from the spectrum of the target ring onto
  its image in the spectrum of the origin ring. This is the case for surjective homomorphisms
  and for localizations, among others.›

theorem (in ring_homo) spectrum_map_homeomorphism:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec" "ideals_extended"
  shows "IsAhomeomorphism(τ⇩t, τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}, g)"
proof-
  have bijg:"g ∈ bij(target_ring.Spec, {f-``u. u∈target_ring.Spec})"
    using spectrum_map_bij assms(3) unfolding g_def by auto
  have hw:"∀u∈target_ring.Spec. f-``u ∈ {f-``u. u∈target_ring.Spec}" by auto
  have contg:"IsContinuous(τ⇩t, τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}, g)"
    using spectrum_map_cont assms(2) hw unfolding g_def by auto
  have opng:"∀U∈τ⇩t. g``U ∈ τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}"
    using spectrum_map_open assms(2,3) unfolding g_def by auto
  have "⋃(τ⇩o{restricted to}{f-``u. u∈target_ring.Spec}) =
    origin_ring.Spec ∩ {f-``u. u∈target_ring.Spec}"
    unfolding top_origin_def RestrictedTo_def using origin_ring.total_spec
    by auto
  then have "⋃(τ⇩o{restricted to}{f-``u. u∈target_ring.Spec}) = {f-``u. u∈target_ring.Spec}"
    using assms(2) by auto moreover
  have "⋃τ⇩t = target_ring.Spec" unfolding top_target_def
    using target_ring.total_spec by auto
  ultimately show ?thesis
    using bij_cont_open_homeo[of g τ⇩t "τ⇩o{restricted to}{f-``u. u∈target_ring.Spec}"]
    bijg contg opng by auto
qed

text‹For a surjective homomorphism the image of the induced map on spectra is the
  closed set defined by the kernel.›

lemma (in ring_homo) spectrum_surj_image:
  assumes "f∈surj(R,S)"
  shows "{f-``u. u∈target_ring.Spec} = V⇩S(ker)"
proof
  have h:"∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
    using spectrum_prime_preimage_surj assms by auto
  then have "∀u∈target_ring.Spec. f-``u ∈ V⇩S(ker)"
    using spectrum_prime_preimage_kernel by auto
  then show "{f-``u. u∈target_ring.Spec} ⊆ V⇩S(ker)" by auto
  have bij:"(λu∈target_ring.Spec. f-``u) ∈ bij(target_ring.Spec, V⇩S(ker))"
    using spectrum_surj_bij assms by auto
  { fix t assume "t ∈ V⇩S(ker)"
    with bij obtain x where x:"x∈target_ring.Spec" "(λu∈target_ring.Spec. f-``u)`x = t"
      unfolding bij_def surj_def by auto
    then have "t = f-``x" using beta by auto
    with x(1) have "t ∈ {f-``u. u∈target_ring.Spec}" by auto
  }
  then show "V⇩S(ker) ⊆ {f-``u. u∈target_ring.Spec}" by auto
qed

text‹The induced map on spectra of a surjective homomorphism sends open sets to open sets
  of the restricted topology.›

lemma (in ring_homo) spectrum_surj_open:
  defines "g ≡ λu∈target_ring.Spec. f-``u"
  assumes "f∈surj(R,S)"
  shows "∀U∈τ⇩t. g``U ∈ τ⇩o {restricted to} V⇩S(ker)"
proof-
  have h:"∀u∈target_ring.Spec. f-``u ∈ origin_ring.Spec"
    using spectrum_prime_preimage_surj assms(2) by auto
  have e:"ideals_extended" using ideals_extended_surj assms(2) by auto
  have "∀U∈τ⇩t. g``U ∈ τ⇩o {restricted to} {f-``u. u∈target_ring.Spec}"
    using spectrum_map_open h e unfolding g_def by auto
  moreover have W:"V⇩S(ker) = {f-``u. u∈target_ring.Spec}"
    using spectrum_surj_image assms(2) by auto
  ultimately show ?thesis unfolding W by simp
qed

text‹A quotient ring has a spectrum homeomorphic
to a closed subspace of the spectrum of the base ring.
Specifically, the closed subspace associated to the
ideal by which we quotient.›

corollary (in ring_homo) surj_homeomorphism:
  assumes "f∈surj(R,S)"
  defines "g ≡ λu∈target_ring.Spec. f -`` u"
  shows "IsAhomeomorphism(τ⇩t, τ⇩o{restricted to}V⇩S(ker), g)"
proof-
  have "⋃(τ⇩o{restricted to}V⇩S(ker)) = origin_ring.Spec ∩ V⇩S(ker)" unfolding
    top_origin_def RestrictedTo_def using origin_ring.total_spec
    by auto
  then have "⋃(τ⇩o{restricted to}V⇩S(ker)) = V⇩S(ker)"
    using origin_ring.closeBasic_def func1_1_L3 f_is_fun
     by auto moreover
  have "⋃τ⇩t = target_ring.Spec" unfolding top_target_def
    using target_ring.total_spec by auto
  ultimately show ?thesis using bij_cont_open_homeo[of g τ⇩t "τ⇩o{restricted to}V⇩S(ker)"]
    spectrum_surj_bij[OF assms(1)] spectrum_surj_open[OF assms(1)]
    spectrum_surj_cont[OF assms(1)]
    unfolding g_def by auto
qed
