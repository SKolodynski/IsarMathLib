section ‹Rings - important localizations›

theory Ring_ZF_7 imports Ring_ZF_6

begin

text‹This theory specializes the localization $S^{-1}R$ of a commutative ring $R$ to two
  important families of multiplicative subsets: the complement $R\setminus P$ of a prime ideal $P$
  (giving the localization $R_P$ at $P$) and the set of powers $\{1,a,a^2,\ldots\}$
  of a single element $a$ (giving the localization $R_a$).›

subsection‹Localization at a prime ideal›

text‹The locale ‹prime_ideal› is a commutative ring together with a prime ideal $P$.›

locale prime_ideal = commutative_ring +
  fixes P
  assumes P_prime: "P◃⇩pR"

text‹The complement of a prime ideal is a subset of the ring.›

lemma (in prime_ideal) complement_subset:
  shows "R - P ⊆ R"
  by auto

text‹The complement of a prime ideal contains the unit, because a prime ideal is not the
  whole ring.›

lemma (in prime_ideal) complement_one:
  shows "𝟭 ∈ R - P"
proof-
  from P_prime have PI:"P◃R" "P ≠ R" unfolding primeIdeal_def by auto
  have "𝟭∈R" using Ring_ZF_1_L2(2) by simp
  moreover have "𝟭 ∉ P"
  proof
    assume "𝟭 ∈ P"
    with PI(1) have "P = R" using ideal_with_one by simp
    with PI(2) show False by simp
  qed
  ultimately show ?thesis by simp
qed

text‹The complement of a prime ideal is closed under multiplication.›

lemma (in prime_ideal) complement_closed:
  shows "(R - P) {is closed under} M"
  unfolding IsOpClosed_def
proof(intro ballI)
  fix x y assume as:"x ∈ R - P" "y ∈ R - P"
  then have "x∈R" "y∈R" by auto
  then have xy:"x⋅y ∈ R" using Ring_ZF_1_L4(3) by simp
  { assume "x⋅y ∈ P"
    with P_prime ‹x∈R› ‹y∈R› have "x∈P ∨ y∈P" using comm_prime_dest by simp
    with as have False by auto
  }
  with xy show "M`⟨x,y⟩ ∈ R - P" by auto
qed

text‹The complement of a prime ideal is a multiplicative subset, so the results about
  localizations apply to it.›

sublocale prime_ideal < prime_loc: mult_subset R A M ringa ringminus ringsub ringm
  ringzero ringone ringtwo ringsq rlistsum rnat_mult listprod pow "R - P"
  using ringAssum commutative complement_subset complement_one complement_closed
  unfolding mult_subset_def mult_subset_axioms_def commutative_ring_def
    commutative_ring_axioms_def ring0_def by auto

text‹The extension of $P$ is a proper ideal of the localization at $P$.›

theorem (in prime_ideal) prime_loc_extension_proper:
  shows "prime_loc.loc_ring.Ideal(ExtendedIdeal(R,M,R - P,P))"
  and "ExtendedIdeal(R,M,R - P,P) ≠ Localization(R,M,R - P)"
proof-
  from P_prime have PI:"P◃R" unfolding primeIdeal_def by simp
  from PI show "prime_loc.loc_ring.Ideal(ExtendedIdeal(R,M,R - P,P))"
    using prime_loc.ext_ideal_is_ideal by simp
  have "P ∩ (R - P) = 0" by auto
  with PI show "ExtendedIdeal(R,M,R - P,P) ≠ Localization(R,M,R - P)"
    using prime_loc.ext_ideal_proper by simp
qed

text‹Every proper ideal of the localization at $P$ is contained in the extension of $P$.
  Together with the previous theorem this shows that the localization at a prime ideal is a
  local ring, whose unique maximal ideal is the extension of $P$.›

theorem (in prime_ideal) prime_loc_local:
  assumes "prime_loc.loc_ring.Ideal(J)" "J ≠ Localization(R,M,R - P)"
  shows "J ⊆ ExtendedIdeal(R,M,R - P,P)"
proof-
  from P_prime have PI:"P◃R" unfolding primeIdeal_def by simp
  have onein:"𝟭 ∈ R - P" using complement_one by simp
  have vsub:"LocalizationMap(R,M,R - P)-``J ⊆ P"
  proof
    fix r assume r:"r ∈ LocalizationMap(R,M,R - P)-``J"
    then have rR:"r∈R" using func1_1_L15 prime_loc.loc_map_fun by auto
    show "r∈P"
    proof(rule ccontr)
      assume "r∉P"
      with rR have "r ∈ R - P" by simp
      with assms(1) r have "J = Localization(R,M,R - P)" using prime_loc.loc_ideal_meets_S by simp
      with assms(2) show False by simp
    qed
  qed
  have ext:"prime_loc.loc_ring.Ideal(ExtendedIdeal(R,M,R - P,P))"
    using prime_loc_extension_proper(1) by simp
  have img:"LocalizationMap(R,M,R - P)``(LocalizationMap(R,M,R - P)-``J) =
    {LocalizationMap(R,M,R - P)`r. r ∈ LocalizationMap(R,M,R - P)-``J}"
    using func_imagedef prime_loc.loc_map_fun func1_1_L3 by auto
  have sub:"LocalizationMap(R,M,R - P)``(LocalizationMap(R,M,R - P)-``J) ⊆
    ExtendedIdeal(R,M,R - P,P)"
  proof
    fix x assume "x ∈ LocalizationMap(R,M,R - P)``(LocalizationMap(R,M,R - P)-``J)"
    with img obtain r where r:"r ∈ LocalizationMap(R,M,R - P)-``J"
      "x = LocalizationMap(R,M,R - P)`r" by auto
    from r(1) vsub have rP:"r∈P" by auto
    from rP PI have rR:"r∈R" using ideal_dest_subset by auto
    from rR r(2) have "x = Fraction(R,M,R - P,r,𝟭)" using prime_loc.loc_map_val by simp
    with rP onein show "x ∈ ExtendedIdeal(R,M,R - P,P)" using prime_loc.ext_ideal_mem by simp
  qed
  show ?thesis using prime_loc.loc_homomorphism.ideals_extended_dest
    prime_loc.localization_ideals_extended assms(1) ext sub by simp
qed

text‹The prime ideals of the localization at $P$ correspond to the prime ideals of $R$ contained
  in $P$.›

theorem (in prime_ideal) prime_loc_prime_image:
  shows "{LocalizationMap(R,M,R - P)-``u. u∈{I∈prime_loc.loc_ring.ideals.
    prime_loc.loc_ring.primeIdeal(I)}} = {q∈ℐ. q◃⇩pR ∧ q ⊆ P}"
proof-
  have "{q∈ℐ. q◃⇩pR ∧ q ∩ (R - P) = 0} = {q∈ℐ. q◃⇩pR ∧ q ⊆ P}"
    by blast
  then show ?thesis using prime_loc.localization_prime_image by simp
qed

subsection‹Localization at the powers of an element›

text‹The powers of an element $a$ of a ring $R$ are the elements $a^n$ for natural numbers $n$,
  where $a^0=1$.›

definition
  "ElementPowers(R,M,c) ≡ {Fold(M,TheNeutralElement(R,M),{⟨k,c⟩. k∈n}). n∈nat}"

text‹The locale ‹ring_element› is a commutative ring together with an element $a$.›

locale ring_element = commutative_ring +
  fixes c
  assumes a_in: "c∈R"

text‹The set of powers of $a$ in terms of the power function of the ring context.›

lemma (in ring_element) powers_eq:
  shows "ElementPowers(R,M,c) = {pow(n,c). n∈nat}"
  unfolding ElementPowers_def by simp

text‹Natural powers of an element are elements of the ring.›

lemma (in ring_element) pow_in_ring:
  assumes "n∈nat"
  shows "pow(n,c) ∈ R"
  using assms a_in mult_monoid.nat_mult_type by simp

text‹The set of powers of $a$ is a subset of the ring.›

lemma (in ring_element) powers_subset:
  shows "ElementPowers(R,M,c) ⊆ R"
  using powers_eq pow_in_ring by auto

text‹The unit is the zeroth power of $a$.›

lemma (in ring_element) powers_one:
  shows "𝟭 ∈ ElementPowers(R,M,c)"
proof-
  have "pow(0,c) = 𝟭" using a_in mult_monoid.nat_mult_zero by simp
  moreover have "pow(0,c)∈{pow(n,c). n∈nat}" by blast
  then have "pow(0,c)∈ElementPowers(R,M,c)" 
    using subst[OF powers_eq[THEN sym], of "λt. pow(0,c)∈t"] by auto
  ultimately show ?thesis by auto
qed

text‹The set of powers of $a$ is closed under multiplication.›

lemma (in ring_element) powers_closed:
  shows "ElementPowers(R,M,c) {is closed under} M"
  unfolding IsOpClosed_def
proof(intro ballI)
  fix x y assume as:"x ∈ ElementPowers(R,M,c)" "y ∈ ElementPowers(R,M,c)"
  from as(1) obtain n where n:"n∈nat" "x = pow(n,c)" using powers_eq by auto
  from as(2) obtain m where m:"m∈nat" "y = pow(m,c)" using powers_eq by auto
  from n(1) m(1) have nm:"n #+ m ∈ nat" by simp
  from n(1) m(1) a_in have A:"pow(n #+ m,c) = pow(n,c)⋅pow(m,c)"
    using mult_monoid.nat_mult_add by simp
  from nm a_in have "pow(n#+m,c)∈{pow(r,c). r∈nat}" by auto
  then have "pow(n#+m,c)∈ElementPowers(R,M,c)" using subst[OF powers_eq[THEN sym],
        of "λt. pow(n#+m,c)∈t"] by auto
  then have "pow(n,c)⋅pow(m,c) ∈ ElementPowers(R,M,c)"
    using subst[OF A, of "λq. q∈ElementPowers(R,M,c)"] by auto
  then show "M`⟨x,y⟩ ∈ ElementPowers(R,M,c)" using n(2) m(2) by force
qed

text‹The set of powers of $a$ is a multiplicative subset, so the results about
  localizations apply to it.›

sublocale ring_element < element_loc: mult_subset R A M ringa ringminus ringsub ringm
  ringzero ringone ringtwo ringsq rlistsum rnat_mult listprod pow "ElementPowers(R,M,c)"
  using ringAssum commutative powers_subset powers_one powers_closed
  unfolding mult_subset_def mult_subset_axioms_def commutative_ring_def
    commutative_ring_axioms_def ring0_def by auto

text‹If a prime ideal contains a power of $a$ then it contains $a$.›

lemma (in ring_element) prime_contains_power:
  assumes "q◃⇩pR"
  shows "∀n∈nat. pow(n,c) ∈ q ⟶ c ∈ q"
proof-
  from assms have qI:"q◃R" "q ≠ R" unfolding primeIdeal_def by auto
  { fix n assume n:"n∈nat"
    have base:"pow(0,c) ∈ q ⟶ c ∈ q"
    proof
      assume "pow(0,c) ∈ q"
      with a_in have "𝟭 ∈ q" using mult_monoid.nat_mult_zero by simp
      with qI(1) have "q = R" using ideal_with_one by simp
      with qI(2) show "c∈q" by simp
    qed
    have step:"∀k∈nat. (pow(k,c) ∈ q ⟶ c ∈ q) ⟶ (pow(k #+ 1,c) ∈ q ⟶ c ∈ q)"
    proof(intro ballI impI)
      fix k assume k:"k∈nat" and ih:"pow(k,c) ∈ q ⟶ c ∈ q" and pq:"pow(k #+ 1,c) ∈ q"
      from k a_in have "pow(k #+ 1,c) = pow(k,c)⋅c"
        using mult_monoid.nat_mult_add_one(1) by simp
      with pq have "pow(k,c)⋅c ∈ q" by simp
      with assms k a_in have "pow(k,c) ∈ q ∨ c ∈ q" using comm_prime_dest pow_in_ring by simp
      with ih show "c∈q" by auto
    qed
    from n base step have "pow(n,c) ∈ q ⟶ c ∈ q" by (rule ind_on_nat1)
  }
  then show ?thesis by simp
qed

text‹A prime ideal is disjoint from the powers of $a$ if and only if it does not contain $a$.›

lemma (in ring_element) prime_disjoint_powers:
  assumes "q◃⇩pR"
  shows "q ∩ ElementPowers(R,M,c) = 0 ⟷ c ∉ q"
proof
  assume dis:"q ∩ ElementPowers(R,M,c) = 0"
  show "c ∉ q"
  proof
    assume aq:"c∈q"
    have "pow(1,c) = c" using a_in mult_monoid.nat_mult_one by simp
    then have "c∈{pow(n,c). n∈nat}" by force
    then have "c ∈ ElementPowers(R,M,c)" 
      using subst[OF powers_eq[THEN sym], of "λq. c:q"] by blast
    with aq dis show False by auto
  qed
next
  assume na:"c ∉ q"
  { fix x assume x:"x ∈ q ∩ ElementPowers(R,M,c)"
    then obtain n where n:"n∈nat" "x = pow(n,c)" using powers_eq by auto
    with x assms have "c∈q" using prime_contains_power by auto
    with na have False by simp
  }
  then show "q ∩ ElementPowers(R,M,c) = 0" by auto
qed

text‹The prime ideals of the localization at the powers of $a$ correspond to the prime ideals
  of $R$ that do not contain $a$.›

theorem (in ring_element) element_loc_prime_image:
  shows "{LocalizationMap(R,M,ElementPowers(R,M,c))-``u.
    u∈{I∈element_loc.loc_ring.ideals. element_loc.loc_ring.primeIdeal(I)}} =
    {q∈ℐ. q◃⇩pR ∧ c ∉ q}"
proof-
  have "{q∈ℐ. q◃⇩pR ∧ q ∩ ElementPowers(R,M,c) = 0} = {q∈ℐ. q◃⇩pR ∧ c ∉ q}"
    using prime_disjoint_powers by auto
  then show ?thesis using element_loc.localization_prime_image by simp
qed
