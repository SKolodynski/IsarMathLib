section ‹Metrization of uniform spaces›

theory MetricUniform_ZF_1 imports MetricUniform_ZF Real_ZF_2

begin

text‹This theory continues ‹MetricUniform_ZF› specializing to the 
  real valued pseudometrics. The ultimate goal is to show that every uniformity 
  is the supremum of a collection of uniformities generated by a pseudometric. In this
  theory we show the AC-free fragments of the proof of this fact.›

subsection‹Defining pseudometrics with weighted chains›

text‹This section describes a method of defining real valued pseudometrics through infimum 
  of the set of weights of all chains connecting two points. 
  The notions of chains and links are defined in the ‹FiniteSeq_ZF› theory, with some 
  properties proven in the ‹Monoid_ZF_1› theory.›

text‹Since we will be using the set of non-zero natural numbers quite a lot, let's
  denote it $\mathbb{N}_1$›

abbreviation positive_nat ("ℕ⇩1") where "ℕ⇩1 ≡ nat∖{0}"

text‹In the ‹monoid1› context we have defined a notion of ‹ChainWeight›.
  The next definition brings that notion into the ‹reals› context›

definition (in reals) RChainWeight where
  "RChainWeight(w,c) ≡ monoid1.ChainWeight(ℝ,Add,w,c)"

text‹Given a chain of elements of $X$ and a weight function 
  $w:X\times X\rightarrow \mathbb{R}$ the weight of a chain is a real number.›

lemma (in reals) chain_weight_real: 
  assumes "n∈nat" "c∈Chains(X,n,x,y)" "w:X×X→ℝ"
  shows "RChainWeight(w,c) ∈ ℝ"
  using assms add_group.monoid.chain_weight_type 
  unfolding RChainWeight_def by simp

text‹Given a weight function $w:X\rightarrow \mathbb{R}^+$ we define 
  the chain distance between two points $x,y\in X$ as the infimum of set of weights of
  all chains of nonzero length connecting $x$ and $y$.›

definition (in reals) ChainDist where
  "ChainDist(X,w,x,y) ≡ inf({RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,x,y))})"

text‹If the weight function is valued in nonnegative real numbers then the weight of every chain
  in nonnegative.›

lemma (in reals) chain_weight_nonneg: 
  assumes "n∈nat" "c∈Chains(X,n,x,y)" "w:X×X→ℝ⇧+"
  shows "RChainWeight(w,c) ∈ ℝ⇧+"
proof -
  from assms have "w O ChainLinks(c) : n→ℝ⇧+"
    using chain_links_fun(2) comp_fun by blast 
  with assms(1) have "(∑(w O ChainLinks(c))) ∈ ℝ⇧+"
    using loop1_valid_in_group3 loop1.sum_nonneg_nonneg by simp
  then show "RChainWeight(w,c) ∈ ℝ⇧+"
    using add_group.monoid1_valid_in_group monoid1.ChainWeight_def
    unfolding RChainWeight_def by simp
qed

text‹If the weight function is nonnegative then the set of weights of all nontrivial chains 
  in $X$ connecting $x$ and $y$ is contained in nonnegative reals.›

corollary (in reals) chain_weights_nonneg: assumes "w:X×X→ℝ⇧+"
  shows "{RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,x,y))} ⊆ ℝ⇧+"
  using assms chain_weight_nonneg by auto

text‹The ‹chain_concat_weight1› lemma from the ‹monoid1› context 
  rewritten using notation used in the ‹reals› context:›

lemma (in reals) rchain_concat_weight:
  assumes "w:X×X→ℝ" "∀x∈X. w`⟨x,x⟩ = 𝟬" "c⇩1∈Chains(X,n⇩1,x,y)" "c⇩2∈Chains(X,n⇩2,y,z)" and
    "n⇩1∈nat" "n⇩2∈nat"  
  shows "RChainWeight(w,Concat(c⇩1,c⇩2)) = RChainWeight(w,c⇩1)\<ra>RChainWeight(w,c⇩2)"
  using assms add_group.monoid1_valid_in_group monoid1.chain_concat_weight1
  unfolding RChainWeight_def by simp

text‹If the weight function is nonnegative then the chain distance is also nonnegative.›

lemma (in reals) chain_dist_nneg: 
  assumes "x∈X" "y∈X" and "w:X×X→ℝ⇧+"
  shows "ChainDist(X,w,x,y) ∈ ℝ⇧+"
  using assms chains_nempty(4) chain_weights_nonneg nneg_has_nneg_inf
  unfolding ChainDist_def by simp

text‹If the weight function is nonnegative and zero on the diagonal then the chain distance 
  obeys the triangle inequality.›

lemma (in reals) chain_dist_tr_ineq: 
  assumes "x∈X" "y∈X" "z∈X" and "w:X×X→ℝ⇧+" "∀x∈X. w`⟨x,x⟩ = 𝟬"
  shows "ChainDist(X,w,x,z) \<lsq> ChainDist(X,w,x,y) \<ra> ChainDist(X,w,y,z)"
proof -
  let ?W⇩1 = "{RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,x,y))}"
  let ?W⇩2 = "{RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,y,z))}"
  let ?W⇩3 = "{RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,x,z))}"
  let ?W⇩4 = "{w⇩1\<ra>w⇩2. ⟨w⇩1,w⇩2⟩ ∈ ?W⇩1×?W⇩2}"
  from R_are_reals assms(1,2,3,4) 
    have "?W⇩1≠∅" "?W⇩2≠∅" "?W⇩3≠∅" "?W⇩4≠∅" "antisym(ROrd)" "w:X×X→ℝ"
    using chains_nempty(4) cart_prod_nempty OrderedGroup_ZF_1_L4E func1_1_L1B 
    unfolding IsAmodelOfReals_def IsAnOrdField_def IsAnOrdRing_def IsLinOrder_def
      by auto
  have "?W⇩4 ⊆ ?W⇩3"
  proof
    fix v assume "v ∈ {w⇩1\<ra>w⇩2. ⟨w⇩1,w⇩2⟩ ∈ ?W⇩1×?W⇩2}"
    then obtain w⇩1 w⇩2 where "w⇩1∈?W⇩1" "w⇩2∈?W⇩2" and "v = w⇩1\<ra>w⇩2"
      by auto
    from ‹w⇩1∈?W⇩1› ‹w⇩2∈?W⇩2› obtain c⇩1 c⇩2 where 
      I: "w⇩1 = RChainWeight(w,c⇩1)"  "w⇩2 = RChainWeight(w,c⇩2)" and 
      II: "c⇩1∈(⋃n∈ℕ⇩1. Chains(X,n,x,y))" "c⇩2∈(⋃n∈ℕ⇩1. Chains(X,n,y,z))"
      by auto
    let ?c⇩3 = "Concat(c⇩1,c⇩2)"
    have "?c⇩3 ∈ (⋃n∈ℕ⇩1. Chains(X,n,x,z))" and "RChainWeight(w,?c⇩3) = v"
    proof -
      from II obtain n⇩1 n⇩2 where "n⇩1∈ℕ⇩1" "n⇩2∈ℕ⇩1" and
        III: "c⇩1 ∈ Chains(X,n⇩1,x,y)" "c⇩2 ∈ Chains(X,n⇩2,y,z)"
        by auto
      from assms(5) ‹n⇩1∈ℕ⇩1› ‹n⇩2∈ℕ⇩1› ‹w:X×X→ℝ› III 
      have "n⇩1∈nat" "n⇩2∈nat" "n⇩1 #+ n⇩2 #+ 1 ∈ ℕ⇩1" and
        "RChainWeight(w,?c⇩3) = RChainWeight(w,c⇩1)\<ra>RChainWeight(w,c⇩2)"
        using rchain_concat_weight by auto
      with ‹v = w⇩1\<ra>w⇩2› I show "RChainWeight(w,?c⇩3) = v" by simp
      from ‹n⇩1∈nat› ‹n⇩2∈nat› III ‹n⇩1 #+ n⇩2 #+ 1 ∈ ℕ⇩1›
      show "?c⇩3 ∈ (⋃n∈ℕ⇩1. Chains(X,n,x,z))"
        using concat_chains(1) by force
    qed
    thus "v∈?W⇩3" by auto
  qed
  from assms(4) ‹?W⇩4 ⊆ ?W⇩3› have "?W⇩1⊆ℝ⇧+" "?W⇩2⊆ℝ⇧+" "?W⇩3⊆ℝ⇧+" 
    using chain_weight_nonneg by auto
  from ‹?W⇩4⊆?W⇩3› ‹?W⇩3⊆ℝ⇧+› have "?W⇩4⊆ℝ⇧+" by blast
  from ‹?W⇩3≠∅› ‹?W⇩3⊆ℝ⇧+› ‹?W⇩4≠∅› ‹?W⇩4⊆ℝ⇧+› ‹antisym(ROrd)› ‹?W⇩4 ⊆ ?W⇩3›
  have "inf(?W⇩3)\<lsq>inf(?W⇩4)" 
    using nneg_has_nneg_inf(1) inf_incl_mono by simp
  with ‹?W⇩1≠∅› ‹?W⇩1⊆ℝ⇧+› ‹?W⇩2≠∅› ‹?W⇩2⊆ℝ⇧+› show ?thesis
    using nonneg_inf_sum_inf(2) unfolding ChainDist_def by simp
qed

text‹If the weight function vanishes on the diagonal then the chain distance
  from $x\in X$ to $x$ is zero.›

lemma (in reals) chain_dist_self_zero: 
  assumes "x∈X" "w:X×X→ℝ⇧+" "w`⟨x,x⟩ = 𝟬"
  shows "ChainDist(X,w,x,x) = 𝟬"
proof -
  let ?c = "{⟨0,x⟩,⟨1,x⟩}"
  from assms(1) have I: "?c ∈ (⋃n∈ℕ⇩1. Chains(X,n,x,x))"
    using chains_nempty(1) by auto  
  from assms have "RChainWeight(w,?c) = 𝟬"
    using OrderedGroup_ZF_1_L4E func1_1_L1B 
      add_group.monoid1_valid_in_group monoid1.weight_one_link
    unfolding RChainWeight_def by simp
  with I have "𝟬∈{RChainWeight(w,c). c∈(⋃n∈ℕ⇩1. Chains(X,n,x,x))}" 
    by blast
  with assms(2) show "ChainDist(X,w,x,x) = 𝟬"
    using chain_weights_nonneg loop1_valid_in_group3 loop1.zero_subset_nneg(4)
    unfolding ChainDist_def by simp
qed

text‹If the weight function is symmetric then the chain distance from $x$ to $y$ is the same
  as from $y$ to $x$.›

lemma (in reals) chain_dist_symm: 
  assumes "w:X×X→ℝ⇧+" "∀x∈X.∀y∈X. w`⟨x,y⟩ = w`⟨y,x⟩"
  shows "ChainDist(X,w,x,y) = ChainDist(X,w,y,x)" 
  using assms basic_props(2) OrderedGroup_ZF_1_L4E func1_1_L1B
      add_group.monoid1_valid_in_group monoid1.weight_set_rev_eq
  unfolding RChainWeight_def ChainDist_def by simp

text‹Given a weight function $w\rightarrow X\times X$ we define the chain weight pseudometric 
  as a function on $X\times X$ that assigns the chain distance to any pair 
  $\langle x,y\rangle \in X\times X$.›

definition (in reals) ChainWeightPMetric where 
  "ChainWeightPMetric(X,w) ≡ {⟨p,ChainDist(X,w,fst(p),snd(p))⟩. p ∈ X×X}"

text‹If $w:X\times X\rightarrow\mathbb{R}^+$ is symmetric and vanishes on the diagonal
  of $X\times X$ then the chain weight pseudometric derived from $w$ 
  is indeed a real-valued pseudometric.›

theorem (in reals) chain_weight_pmetric:
  assumes "w:X×X→ℝ⇧+" "∀x∈X. w`⟨x,x⟩ = 𝟬" "∀x∈X.∀y∈X. w`⟨x,y⟩ = w`⟨y,x⟩"
  shows "IsApseudoMetric(ChainWeightPMetric(X,w),X,ℝ,Add,ROrd)"
  using assms ZF_fun_from_total chain_dist_nneg ZF_fun_from_tot_val1 
    chain_dist_self_zero chain_dist_symm chain_dist_tr_ineq
  unfolding ChainWeightPMetric_def IsApseudoMetric_def by auto

subsection‹Metrization lemma›

text‹The metrization theorem states that a uniform space is pseudometrizable if and only if 
  its uniformity has an countable base. In this sectiopn we show the key lemma that leads to 
  the metrization theorem.

  In the literature the metrization lemma (proven below) is usually stated as asserting
  existence of a pseudometric satisfying certain conditions. We prefer here a more explicit
  approach where we specify the construction of that pseudometric as a series of definitions,
  then show that the resulting function indeed satisfies the required conditions.
  The drawback of this approach is that we have to come up with names for the intermediate objects
  representing stages of that construction. Naming is hard, so as usual if the reader has an idea
  for better names please let me know.›

text‹We define the ‹DistWeight(𝒰,p)› expression as equal to (real) $0$ if 
  $p\in\bigcap_{n\in\mathbb{N}} \mathcal{U}_n$ or otherwise equal to $(\frac{1}{2})^n$, where
  $n$ is the only natural number such that $p\in\mathcal{U}_n\setminus\mathcal{U}_{n+1}$.
  To understand the context in which this definition is intended to be used suppose 
  $\mathcal{U}:\mathbb{N}\rightarrow\mathcal{P}(X\times X)$ is a sequence of subsets 
  of $X\times X$ such that $\mathcal{U}_{n+1}\subseteq \mathcal{U}_n$ for $n\in\mathbb{N}$. 
  Then we have the identity 
  $\bigcup_{n\in \mathbb{N}} (\mathcal{U}_n\setminus\mathcal{U}_{n+1})=\mathcal{U}_0\setminus \bigcap_{n\in \mathbb{N}} \mathcal{U}_n$ 
  and for any $p\in\bigcup_{n\in \mathbb{N}} (\mathcal{U}_n\setminus \mathcal{U}_{n+1})$ 
  there is exactly one $n\in \mathbb{N}$ such that $p\in \mathcal{U}_n\setminus \mathcal{U}_{n+1}$
  This means that the set $\{n\in n\in \mathbb{N}: p\in \mathcal{U}_n\setminus \mathcal{U}_{n+1}\}$ 
  is a singleton and we can use the $\bigcup \{ x\} = x$ identity to extract its only element.›

definition (in reals) DistWeight where 
  "DistWeight(𝒰,p) ≡ if p ∈ (⋂n∈nat. 𝒰`(n)) then 𝟬 else pow(⋃{n∈nat. p∈(𝒰`(n)∖𝒰`(n #+ 1))},½)"

text‹Suppose $\mathcal{U}:\mathbb{N}\rightarrow \mathcal{P}(X\times X)$ is a sequence of subsets 
  of $X\times X$ such that $\mathcal{U}_{n+1}\subseteq \mathcal{U}_n$ 
  for all $n\in\mathbb{N}$.
  Assume further that $p\in\mathcal{U}_n\setminus \mathcal{U}_{n+1}$ for some $n\in\mathbb{N}$.
  Then ‹DistWeight(𝒰,p)› is equal to $\left(\frac{1}{2}\right)^n$.›

lemma (in reals) dist_weight_val: 
  assumes "𝒰:nat→Pow(X×X)" "∀i∈nat. 𝒰`(i #+ 1) ⊆ 𝒰`(i)" and
    "n∈nat" "p ∈ (𝒰`(n)∖𝒰`(n #+ 1))" 
  shows "DistWeight(𝒰,p) = pow(n,½)"
proof -
  let ?𝒱 = "{⟨i,𝒰`(i)∖𝒰`(i #+ 1)⟩. i∈nat}"
  have I: "∀i∈nat. ?𝒱`(i) = 𝒰`(i)∖𝒰`(i #+ 1)" 
    using ZF_fun_from_tot_val2 by simp
  from assms(1,2) have 
    "?𝒱:nat→Pow(X×X)" "?𝒱 {is pairwise disjoint}" and 
    II: "(⋃n∈nat. 𝒰`(n)∖𝒰`(n #+ 1)) = 𝒰`(0)∖(⋂n∈nat. 𝒰`(n))"
    using decr_pair_disj1(1,3,4) by simp_all
  from assms(3,4) I have "p∈?𝒱`(n)" by simp
  with assms(3) ‹?𝒱:nat→Pow(X×X)› ‹?𝒱 {is pairwise disjoint}›
  have "n = ⋃{i∈nat. p∈?𝒱`(i)}" using get_the_one by simp
  with assms(3,4) I have "n = ⋃{i∈nat. p∈𝒰`(i)∖𝒰`(i #+ 1)}" and 
    "p ∈ (⋃n∈nat. 𝒰`(n)∖𝒰`(n #+ 1))"
    by auto
  with II show ?thesis unfolding DistWeight_def by simp
qed
